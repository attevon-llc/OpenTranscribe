# Test-runner tooling ONLY — the plugins pytest needs to execute this repo's suite.
#
# Split out of requirements-dev.txt (issue #577) so a container can become
# test-capable without also pulling the linter/formatter/mutation/browser stack.
# `requirements-dev.txt` includes this file, so the pins live in exactly one place
# and the venv and any test image cannot drift apart the way #492 describes.
#
# Deliberately NOT here: pytest-cov, mutmut, pytest-playwright, ruff/mypy/black.
# Those are developer-workstation tools; baking them into a production-adjacent
# image adds attack surface for something no test run in that image needs.
#
# Install (image): pip install --user -r requirements-test.txt  — see Dockerfile.test.
# Install (venv):  covered by requirements-dev.txt.

pytest==9.1.1
pytest-asyncio==1.4.0  # For testing async code
pytest-xdist==3.8.0  # Parallel execution — pyproject addopts uses "-n auto"
# REQUIRED, not optional: pyproject addopts passes --timeout, so a missing plugin is a
# collection error, not a silently skipped feature. The suite serialises DDL tests behind a
# Postgres advisory lock; without a timeout a genuine lock cycle hangs the run forever
# instead of failing (issue #431).
pytest-timeout==2.4.0
